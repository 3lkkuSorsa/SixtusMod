fire_ministers = {
	custom_effect_tooltip = GER_fire_all_ministers_tt
	hidden_effect = {
		THIS = {
			every_character = {
				limit = {
					OR = {
						is_character_slot = minister
						is_character_slot = head_of_government
					}
				}
				owner = {
					deactivate_advisor = PREV
				}

				remove_trait = {
					character = THIS
					slot = minister
					trait = interior_minister
				}
				remove_trait = {
					character = THIS
					slot = minister
					trait = foreign_minister
				}
				remove_trait = {
					character = THIS
					slot = minister
					trait = finance_minister
				}
				remove_trait = {
					character = THIS
					slot = minister
					trait = economy_minister
				}
				remove_trait = {
					character = THIS
					slot = minister
					trait = army_minister
				}
			}
		}
	}
}
