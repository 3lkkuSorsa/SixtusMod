GER_spring_coalition_category = {
	GER_elections_timer = {
		allowed = {
			original_tag = GER
		}

		available = {
			always = no
		}

		activation = {
			always = no
		}

		days_mission_timeout = 265

		priority = 10

		timeout_effect = {
			country_event = {
				id = GER_elections_36.1
			}
		}
	}

	GER_zentrum_ambitions = {
		allowed = {
			original_tag = GER
		}

		available = {
			has_completed_focus = GER_political_manoeuvres
		}

		activation = {
			always = no
		}

		days_mission_timeout = 120

		timeout_effect = {
			country_event = {
				id = GER_political_manoeuvres.2
				days = 5
			}
		}
	}

	GER_kampf_aus_der_mitte_mission = {
		allowed = {
			original_tag = GER
		}

		available = {
			custom_trigger_tooltip = {
				tooltip = GER_kampf_aus_der_mitte_available_tt
				check_variable = {
					var = GER.coalition_stability
					value = 9
					compare = greater_than
				}				
			}
		}

		activation = {
			always = no
		}

		days_mission_timeout = 40

		selectable_mission = yes

		is_good = no

		priority = 9

		timeout_effect = {
			if = {
				limit = {
					check_variable = {
						var = GER.centrists_for_left
						value = 25
						compare = less_than
					}
				}
				activate_mission = GER_kampf_aus_der_mitte_mission
				set_temp_variable = {
					var = GER.coalition_stability_temp
					value = GER.coalition_stability
				}
				multiply_temp_variable = {
					var = GER.coalition_stability_temp
					value = -1
				}
				add_to_variable = {
					var = GER.centrists_for_left
					value = GER.coalition_stability_temp
					tooltip = GER_coalition_centrists_for_left_tt
				}
				else = {
					country_event = {
						id = GER_zentrum_fails.1
						days = 5
					}
				}
				if = {
					limit = {
						set_temp_variable = {
							var = GER.centrists_for_left_temp
							value = GER.centrists_for_left
						}
						set_temp_variable = {
							var = GER.coalition_stability_temp
							value = GER.coalition_stability
						}
						multiply_temp_variable = {
							var = GER.coalition_stability_temp
							value = -1
						}
						add_to_temp_variable = {
							var = GER.centrists_for_left_temp
							value = GER.coalition_stability_temp
						}
						check_variable = {
							var = GER.centrists_for_left_temp
							value = 25
							compare = greater_than_or_equals
						}
					}
					custom_effect_tooltip = GER_kampf_aus_der_mitte_crisis_tt
				}
			}
			if = {
				limit = {
					has_completed_focus = GER_repeal_anti_trade_union_legislation 
					has_completed_focus = GER_expand_worker_protections 
					has_completed_focus = GER_imperial_mediation
					has_country_flag = GER_schleicher_coup_planned
				}
				custom_effect_tooltip = GER_kampf_aus_der_mitte_success_tt
				add_to_variable = {
					var = GER.coalition_stability
					value = 1
					tooltip = GER_coalition_stability_change_tt
				}
				else = {
					random_list = {
						65 = {
							add_to_variable = {
								var = GER.coalition_stability
								value = -1
								tooltip = GER_coalition_stability_change_tt
							}
						}
						35 = {
							add_to_variable = {
								var = GER.coalition_stability
								value = -2
								tooltip = GER_coalition_stability_change_tt
							}
						}
					}
				}
			}
		}
		complete_effect = {
			country_event = {
				id = GER_political_manoeuvres.5
				days = 2
			}
			remove_ideas = GER_kampf_aus_der_mitte_idea
		}
	}

	GER_address_the_ongoing_crisis = {
		allowed = {
			original_tag = GER
		}

		available = {
			has_completed_focus = GER_reconcile_volksblock
		}

		activation = {
			has_idea = GER_government_in_crisis
		}

		days_mission_timeout = 35

		timeout_effect = {
			country_event = {
				id = GER_zentrum_fails.2
				days = 5
			}
		}

		complete_effect = {
			remove_ideas = GER_government_in_crisis
		}
	}
}
