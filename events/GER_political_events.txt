		trigger = {
			has_idea = GER_schleicher_in_prussia

				has_idea = GER_prussian_grand_coalition
			}
		}
	}

	option = {
		name = "Weimar majority!"
		add_stability = 0.07
		trigger = {
			has_country_flag = GER_ensured_coalition_victory
			has_idea = GER_prussian_grand_coalition
			NOT = {
				has_idea = GER_schleicher_in_prussia
			}
		}
	}

	option = {
		name = "DkP establishes a right-wing coalition!"
		add_stability = -0.07
		if = {
			limit = {
				has_idea = GER_prussian_grand_coalition
			}
			swap_ideas = {
				remove_idea = GER_prussian_grand_coalition
				add_idea = GER_conservative_prussian_government
			}
			else = {
				swap_ideas = {
					remove_idea = GER_minority_prussian_government
					add_idea = GER_conservative_prussian_government
				}
			}
		}
		trigger = {
			NOT = {
				OR = {
					has_country_flag = GER_ensured_coalition_victory
					has_idea = GER_schleicher_in_prussia
				}
			}
		}
	}
}

country_event = {
	id = GER_elections_36.1
	title = "Election results are in!"
	desc = GER_elections_36.1.d

	is_triggered_only = yes

	immediate = {
		hidden_effect = {
			remove_mission = GER_elections_timer
			clr_country_flag = GER_elections_36
		}
	}

	option = {
		name = "Victory for the Spring Coalition!"
		trigger = {
			has_country_flag = GER_spring_coalition
		}
		set_country_flag = GER_spring_coalition_2
		country_event = {
			id = GER_spring_coalition.4
			days = 4
		}
		unlock_national_focus = GER_promised_and_delivered
		if = {
			limit = {
				has_country_flag = GER_agreed_to_stegerwald
			}
			promote_character = {
				character = GER_adam_stegerwald
			}
			set_politics = {
				ruling_party = conservatism
			}
			hidden_effect = {
				add_ideas = GER_adam_stegerwald_hog_idea
			}
		}
		if = {
			limit = {
				has_country_flag = GER_agreed_to_bruning
			}
			promote_character = {
				character = GER_heinrich_bruning
			}
			set_politics = {
				ruling_party = conservatism
			}
			hidden_effect = {
				add_ideas = GER_heinrich_bruning_hog_idea
			}
		}
		hidden_effect = {
			remove_ideas = GER_kuno_von_westarp_hog_idea
		}
	}

	option = {
		name = "Volksblock triumphant!"
		trigger = {
			NOT = {
				has_country_flag = GER_spring_coalition
			}
		}
		if = {
			limit = {
				check_variable = {
					var = z_votes
					value = spd_votes
					compare = greater_than_or_equals
				}
			}
			custom_effect_tooltip = GER_Zentrum_wins_tt
			set_country_flag = GER_winner_Z_volksblock
			else = {
				custom_effect_tooltip = GER_SPD_wins_tt
				set_country_flag = GER_winner_SPD_volksblock
			}
		}
		clr_country_flag = GER_show_moderate_tree
		set_country_flag = GER_show_democratic_tree
		mark_focus_tree_layout_dirty = yes
		unlock_national_focus = GER_volksblock_triumphs
		remove_ideas = GER_spring_turns_into_winter
	}
}
