add_namespace = GER_volksblock_cabinet

country_event = {
	id = GER_volksblock_cabinet.0
	title = "Question of the chancellorship"
	desc = GER_zentrum_fails.1.d

	is_triggered_only = yes

	option = {
		name = "The winner shall get the chancellorship, of course!"
		custom_effect_tooltip = GER_canidate_choice_tt_volksblock
		if = {
			limit = {
				has_country_flag = GER_winner_SPD_volksblock
			}
			set_politics = {
				ruling_party = social_democracy
			}
			else = {
				set_politics = {
					ruling_party = conservatism
				}
			}
		}
	}

	option = {
		name = "Maybe we should appoint a liberal compromise candidate?"
		custom_effect_tooltip = GER_canidate_choice_tt_volksblock
		set_politics = {
			ruling_party = social_liberalism
		}
		promote_character = {
			character = GER_otto_gessler
		}
		add_to_variable = {
			var = GER.fvp_cab_seats
			value = -1
			tooltip = GER_cabinet_seat_change_fvp_tt
		}
		if = {
			limit = {
				has_country_flag = GER_winner_SPD_volksblock
			}
			add_to_variable = {
				var = GER.spd_cab_seats
				value = 1
				tooltip = GER_cabinet_seat_change_spd_tt
			}
			else = {
				set_politics = {
					ruling_party = conservatism
				}
				add_to_variable = {
					var = GER.zentrum_cab_seats
					value = 1
					tooltip = GER_cabinet_seat_change_zentrum_tt
				}
			}
		}
	}
}
