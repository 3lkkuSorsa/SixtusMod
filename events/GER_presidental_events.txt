add_namespace = GER_death_of_ebert

country_event = {
	id = GER_death_of_ebert.1
	title = "President Ebert falls seriously ill"
	desc = GER_death_of_ebert.1.d

	is_triggered_only = yes
	fire_only_once = yes

	option = {
		name = "Hopefully he recovers soon."
		hidden_effect = {
			country_event = {
				id = GER_death_of_ebert.2
				days = 15
			}
		}
	}
}

country_event = {
	id = GER_death_of_ebert.2
	title = "President Ebert passes away"
	desc = GER_death_of_ebert.2.d

	is_triggered_only = yes
	fire_only_once = yes

	option = {
		name = "A great deal of tragedy! Appoint Luther as an interim president."
		init_elections_25 = yes
		activate_mission = GER_first_round_1925
	}
}

country_event = {
	id = GER_death_of_ebert.3
	title = "First round of the 1925 presidential election"
	desc = GER_death_of_ebert.3.d

	is_triggered_only = yes
	fire_only_once = yes

	option = {
		name = "No candidate wins a absolute majority."
		init_elections_25_round_2 = yes
		activate_mission = GER_second_round_1925
		hidden_effect = {
			country_event = {
				id = GER_death_of_ebert.4
				days = 10
			}
			country_event = {
				id = GER_death_of_ebert.5
				days = 15
			}
		}
	}
}

	id = GER_death_of_ebert.4
	title = "Right-wing united behind Paul von Hindenburg"
	desc = GER_death_of_ebert.4.d

	is_triggered_only = yes
	fire_only_once = yes

	immediate = {
		hidden_effect = {
			clear_array = candidates_array

		name = "But he isn't even a politician!"
	}
}
